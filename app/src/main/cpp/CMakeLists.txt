cmake_minimum_required(VERSION 3.22)
project(physical_nlu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Optimize C kernels as well as C++ even in a debuggable APK.
add_compile_options(-O3)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third-party/llama.cpp ${CMAKE_BINARY_DIR}/llama)
add_library(physical_nlu SHARED nlu_jni.cpp)
target_link_libraries(physical_nlu PRIVATE llama android log)
target_compile_options(physical_nlu PRIVATE -O3)
target_link_options(physical_nlu PRIVATE "-Wl,-z,max-page-size=16384")
